1. Authentication Flow
1.1 User Login
javascript
// Example: Login request
const login = async (email, password) => {
  const response = await fetch('/auth/login', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify({email, password})
  });
  
  const data = await response.json();
  // Store token
  localStorage.setItem('auth_token', data.access_token);
  localStorage.setItem('refresh_token', data.refresh_token);
  
  return data;
};
1.2 Token Management
javascript
// Interceptor for adding token to requests
const apiClient = axios.create({
  baseURL: 'http://your-api-url.com'
});

apiClient.interceptors.request.use(config => {
  const token = localStorage.getItem('auth_token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// Handle token refresh
apiClient.interceptors.response.use(
  response => response,
  async error => {
    if (error.response.status === 401) {
      // Refresh token logic
      await refreshToken();
      // Retry original request
      return apiClient(error.config);
    }
    return Promise.reject(error);
  }
);

2.2 API Service Layer
javascript
// api/suggestions.js
export const calculateRange = async (params) => {
  const queryParams = new URLSearchParams({
    latitude: params.latitude,
    longitude: params.longitude,
    car_id: params.carId,
    battery_percentage: params.batteryPercentage,
    terrain_factor: params.terrainFactor || 1.0,
    weather_factor: params.weatherFactor || 1.0
  });
  
  const response = await fetch(
    `/suggestions/range-calculation?${queryParams}`,
    {
      headers: {
        'Authorization': `Bearer ${localStorage.getItem('auth_token')}`
      }
    }
  );
  
  if (!response.ok) {
    throw new Error('Range calculation failed');
  }
  
  return await response.json();
};

5. Error Handling in UI
5.1 Error Boundary Component
javascript
// components/ErrorBoundary.jsx
class ErrorBoundary extends React.Component {
  constructor(props) {
    super(props);
    this.state = { hasError: false, error: null };
  }
  
  static getDerivedStateFromError(error) {
    return { hasError: true, error };
  }
  
  render() {
    if (this.state.hasError) {
      return (
        <div className="error-container">
          <h3>Something went wrong</h3>
          <p>{this.state.error.message}</p>
          <button onClick={() => window.location.reload()}>
            Reload Application
          </button>
        </div>
      );
    }
    
    return this.props.children;
  }
}
5.2 API Error Handling
javascript
// utils/apiHelpers.js
export const handleApiError = (error) => {
  if (error.response) {
    switch (error.response.status) {
      case 401:
        return 'Session expired. Please login again.';
      case 403:
        return 'You do not have permission to access this resource.';
      case 404:
        return 'Resource not found.';
      case 500:
        return 'Server error. Please try again later.';
      default:
        return error.response.data?.detail || 'An error occurred';
    }
  } else if (error.request) {
    return 'Network error. Please check your connection.';
  } else {
    return error.message || 'An unexpected error occurred';
  }
};

7. State Management (Redux/Context)
7.1 Context API Example
javascript
// context/RangeContext.jsx
import React, { createContext, useContext, useReducer } from 'react';

const RangeContext = createContext();

const rangeReducer = (state, action) => {
  switch (action.type) {
    case 'SET_RANGE_DATA':
      return { ...state, rangeData: action.payload };
    case 'SET_LOADING':
      return { ...state, loading: action.payload };
    case 'SET_ERROR':
      return { ...state, error: action.payload };
    default:
      return state;
  }
};

export const RangeProvider = ({ children }) => {
  const [state, dispatch] = useReducer(rangeReducer, {
    rangeData: null,
    loading: false,
    error: null
  });
  
  return (
    <RangeContext.Provider value={{ state, dispatch }}>
      {children}
    </RangeContext.Provider>
  );
};

export const useRange = () => useContext(RangeContext);
